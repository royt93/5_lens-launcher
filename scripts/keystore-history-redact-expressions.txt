# git-filter-repo --replace-text expressions for SEC-001.
#
# `git filter-repo --path app/keystore.jks --invert-paths` (the command in
# doc/task/SEC-001_OWNER_CHECKLIST.md step 5) only drops the keystore.jks BLOB from
# history. It does NOT touch gradle.properties: that file is still tracked today (with
# clean content), so filter-repo keeps every historical revision of it — including the
# ones where KS_ALIAS/KS_PW were committed in plaintext (commits a8701ba, ac7e944,
# 67e7f81; see `git log --all -p -- gradle.properties`). These expressions redact that
# text from every commit. Regenerate with fresh values if the exposed credential
# strings ever change.
literal:KS_ALIAS=mckimquyen==>KS_ALIAS=REDACTED
literal:KS_ALIAS=loi==>KS_ALIAS=REDACTED
literal:KS_PW=27072000==>KS_PW=REDACTED
literal:KS_PW=04021993==>KS_PW=REDACTED
